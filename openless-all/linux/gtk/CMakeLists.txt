cmake_minimum_required(VERSION 3.20)
project(openless-gtk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4>=4.12 webkitgtk-6.0 libsoup-3.0)
pkg_check_modules(LAYER QUIET IMPORTED_TARGET gtk4-layer-shell-0)
add_executable(openless-gtk src/main.cpp)
target_compile_options(openless-gtk PRIVATE -Wall -Wextra -Werror)
target_link_libraries(openless-gtk PRIVATE PkgConfig::GTK)
if(LAYER_FOUND)
  target_compile_definitions(openless-gtk PRIVATE OPENLESS_GTK_LAYER_SHELL=1)
  target_link_libraries(openless-gtk PRIVATE PkgConfig::LAYER)
else()
  message(WARNING "gtk4-layer-shell unavailable: GTK main window works, Wayland overlays are disabled; Qt remains the default")
endif()
install(TARGETS openless-gtk RUNTIME DESTINATION lib/openless)
